# Melkino — تنظیمات پایه‌ی امنیتی Apache
Options -Indexes

# مینی‌اپ بله (نسخهٔ وب) صفحه را iframe می‌کند.
# پیام «This content is blocked» یعنی X-Frame-Options یا CSP قدیمی
# هنوز در پاسخ هست. اول هر دو را پاک می‌کنیم، بعد CSP درست را می‌گذاریم.
# اگر unset نشود، مرورگر چند CSP را با هم قطع می‌دهد و باز هم مسدود می‌ماند.
<IfModule mod_headers.c>
	Header unset X-Frame-Options
	Header always unset X-Frame-Options
	Header unset Content-Security-Policy
	Header always unset Content-Security-Policy
	Header always set Content-Security-Policy "frame-ancestors 'self' https://*.bale.ai https://bale.ai https://web.bale.ai https://ble.ir https://*.telegram.org https://web.telegram.org https://*.eitaa.com https://web.eitaa.com https://eitaa.com"
</IfModule>

# فشرده‌سازی خروجی متنی (HTML پنل ادمین ~۲۴۵KB است؛ با gzip به ~۴۰KB
# می‌رسد و صفحه‌ها روی هاست اشتراکی/موبایل خیلی سریع‌تر بالا می‌آیند)
<IfModule mod_deflate.c>
	AddOutputFilterByType DEFLATE text/html text/plain text/css text/javascript application/javascript application/json application/xml image/svg+xml
</IfModule>
<IfModule mod_filter.c>
	AddOutputFilterByType DEFLATE text/html text/plain text/css text/javascript application/javascript application/json application/xml image/svg+xml
</IfModule>

# فایل‌های حساس هرگز از وب سرو نشوند (با PHP خوانده می‌شوند، نه مرورگر)
# نکته: theme.json عمداً اینجا نیست چون تب تمِ پنل ادمین آن را مستقیم fetch می‌کند
<FilesMatch "^(config\.secrets.*|config\.json|onboarding-logo\.json|\.env)$">
	Require all denied
</FilesMatch>
<IfModule !mod_authz_core.c>
	<FilesMatch "^(config\.secrets.*|config\.json|onboarding-logo\.json|\.env)$">
		Order allow,deny
		Deny from all
	</FilesMatch>
</IfModule>

# لاگ‌ها و فایل‌های پشتیبان هرگز سرو نشوند
<FilesMatch "\.(log|sql|zip|tar|gz|bak)$">
	Require all denied
</FilesMatch>

# فایل‌های JS/CSS هر بار اعتبارسنجی شوند تا پس از آپدیت کد، مرورگرِ
# ادمین نسخه‌ی کهنه را اجرا نکند (کشِ InfinityFree هم دور زده می‌شود)
<IfModule mod_headers.c>
	<FilesMatch "\.(js|css)$">
		Header set Cache-Control "no-cache, must-revalidate"
	</FilesMatch>
</IfModule>
<IfModule !mod_authz_core.c>
	<FilesMatch "\.(log|sql|zip|tar|gz|bak)$">
		Order allow,deny
		Deny from all
	</FilesMatch>
</IfModule>
